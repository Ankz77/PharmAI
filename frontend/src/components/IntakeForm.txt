import React, { useState } from 'react';

const CATEGORIZED_SYMPTOMS = {
    "General & Vitals": ["fever", "fatigue", "weight loss", "sweating", "chills"],
    "Neurological": ["headache", "dizziness", "confusion", "weakness"],
    "Respiratory": ["cough", "breathlessness", "chest pain", "runny nose"],
    "Gastrointestinal": ["vomiting", "nausea", "abdominal pain", "diarrhoea", "constipation"]
};

export default function IntakeForm({ onSubmit, loading }) {
    const [formData, setFormData] = useState({
        age: 25,
        gender: 'Female',
        weight_kg: 70.0,
        temperature: 37.2,
        symptom_duration_days: 3,
        is_pregnant: false,
        symptoms: ['fever', 'headache', 'fatigue']
    });

    const [searchQuery, setSearchQuery] = useState('');

    const allSymptoms = Object.values(CATEGORIZED_SYMPTOMS).flat();
    const filteredSymptoms = searchQuery.trim() === '' 
        ? [] 
        : allSymptoms.filter(s => s.toLowerCase().includes(searchQuery.toLowerCase()) && !formData.symptoms.includes(s));

    const handleChange = (e) => {
        const { name, value, type, checked } = e.target;
        setFormData({
            ...formData,
            [name]: type === 'checkbox' ? checked : value
        });
    };

    const handleToggleSymptom = (symptom) => {
        const updatedSymptoms = formData.symptoms.includes(symptom)
            ? formData.symptoms.filter(s => s !== symptom)
            : [...formData.symptoms, symptom];

        setFormData({
            ...formData,
            symptoms: updatedSymptoms
        });
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        const payload = {
            ...formData,
            age: parseInt(formData.age, 10),
            weight_kg: parseFloat(formData.weight_kg),
            temperature: formData.temperature ? parseFloat(formData.temperature) : null,
            symptom_duration_days: parseInt(formData.symptom_duration_days, 10),
            allergies: [],
            chronic_conditions: [],
            symptoms: formData.symptoms
        };
        onSubmit(payload);
    };

    return (
        <form onSubmit={handleSubmit} className="bg-white p-8 rounded-xl shadow-lg border border-slate-200 space-y-6">
            {/* Header Section */}
            <div className="border-b border-slate-200 pb-4">
                <h2 className="text-2xl font-extrabold text-blue-900 tracking-tight">Patient Clinical Intake</h2>
                <p className="text-xs text-slate-500 mt-1">Enter patient vitals and select presenting symptoms for evaluation.</p>
            </div>
            
            {/* Vitals Grid */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-5">
                <div>
                    <label className="block text-xs font-semibold uppercase tracking-wider text-slate-700 mb-1">Age (Years)</label>
                    <input 
                        type="number" 
                        name="age" 
                        value={formData.age} 
                        onChange={handleChange} 
                        className="w-full border border-slate-300 rounded-lg p-2.5 text-slate-800 bg-slate-50 focus:bg-white focus:ring-2 focus:ring-teal-600 focus:border-teal-600 outline-none transition" 
                        required 
                    />
                </div>
                <div>
                    <label className="block text-xs font-semibold uppercase tracking-wider text-slate-700 mb-1">Biological Sex</label>
                    <select 
                        name="gender" 
                        value={formData.gender} 
                        onChange={handleChange} 
                        className="w-full border border-slate-300 rounded-lg p-2.5 text-slate-800 bg-slate-50 focus:bg-white focus:ring-2 focus:ring-teal-600 focus:border-teal-600 outline-none transition"
                    >
                        <option value="Female">Female</option>
                        <option value="Male">Male</option>
                    </select>
                </div>
                <div>
                    <label className="block text-xs font-semibold uppercase tracking-wider text-slate-700 mb-1">Weight (kg)</label>
                    <input 
                        type="number" 
                        step="0.1" 
                        name="weight_kg" 
                        value={formData.weight_kg} 
                        onChange={handleChange} 
                        className="w-full border border-slate-300 rounded-lg p-2.5 text-slate-800 bg-slate-50 focus:bg-white focus:ring-2 focus:ring-teal-600 focus:border-teal-600 outline-none transition" 
                        required 
                    />
                </div>
                <div>
                    <label className="block text-xs font-semibold uppercase tracking-wider text-slate-700 mb-1">Body Temp (°C)</label>
                    <input 
                        type="number" 
                        step="0.1" 
                        name="temperature" 
                        value={formData.temperature} 
                        onChange={handleChange} 
                        className="w-full border border-slate-300 rounded-lg p-2.5 text-slate-800 bg-slate-50 focus:bg-white focus:ring-2 focus:ring-teal-600 focus:border-teal-600 outline-none transition" 
                    />
                </div>
                <div>
                    <label className="block text-xs font-semibold uppercase tracking-wider text-slate-700 mb-1">Duration (Days)</label>
                    <input 
                        type="number" 
                        name="symptom_duration_days" 
                        value={formData.symptom_duration_days} 
                        onChange={handleChange} 
                        className="w-full border border-slate-300 rounded-lg p-2.5 text-slate-800 bg-slate-50 focus:bg-white focus:ring-2 focus:ring-teal-600 focus:border-teal-600 outline-none transition" 
                        required 
                    />
                </div>
                {formData.gender === 'Female' && (
                    <div className="flex items-center pt-6">
                        <label className="inline-flex items-center cursor-pointer space-x-2">
                            <input 
                                type="checkbox" 
                                name="is_pregnant" 
                                checked={formData.is_pregnant} 
                                onChange={handleChange} 
                                className="h-5 w-5 text-teal-600 border-slate-300 rounded focus:ring-teal-500 accent-teal-600" 
                            />
                            <span className="text-sm font-semibold text-slate-700">Patient is Pregnant</span>
                        </label>
                    </div>
                )}
            </div>

            {/* Combined Symptom Selection Component */}
            <div className="space-y-4 border-t border-slate-200 pt-5">
                <div className="flex items-center justify-between">
                    <label className="text-sm font-bold text-blue-900 uppercase tracking-wider">Presenting Symptoms</label>
                    <span className="text-xs font-semibold text-teal-700 bg-teal-50 px-2.5 py-1 rounded-full border border-teal-200">
                        {formData.symptoms.length} Selected
                    </span>
                </div>
                
                {/* Type-Ahead Search Dropdown */}
                <div className="relative">
                    <input 
                        type="text" 
                        placeholder="Type to search symptoms (e.g. headache, fever)..." 
                        value={searchQuery}
                        onChange={(e) => setSearchQuery(e.target.value)}
                        className="w-full p-3 border border-slate-300 rounded-lg text-sm text-slate-800 bg-white placeholder-slate-400 focus:ring-2 focus:ring-teal-600 focus:border-teal-600 outline-none shadow-sm transition"
                    />
                    {filteredSymptoms.length > 0 && (
                        <ul className="absolute z-20 w-full bg-white border border-slate-200 mt-1 rounded-lg shadow-xl max-h-48 overflow-y-auto divide-y divide-slate-100">
                            {filteredSymptoms.map((symptom) => (
                                <li 
                                    key={symptom}
                                    onClick={() => {
                                        handleToggleSymptom(symptom);
                                        setSearchQuery('');
                                    }}
                                    className="p-3 hover:bg-teal-50 hover:text-teal-900 cursor-pointer capitalize text-sm font-medium text-slate-700 transition"
                                >
                                    + Add <span className="font-bold">{symptom}</span>
                                </li>
                            ))}
                        </ul>
                    )}
                </div>

                {/* Selected Symptoms Active Tray */}
                <div className="bg-slate-50 p-3.5 rounded-lg border border-slate-200">
                    <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider block mb-2">Active Selection Tray:</span>
                    <div className="flex flex-wrap gap-2 min-h-[36px]">
                        {formData.symptoms.length === 0 ? (
                            <span className="text-sm text-slate-400 italic">No symptoms selected. Select from categories below or search above.</span>
                        ) : (
                            formData.symptoms.map((symptom) => (
                                <span 
                                    key={symptom} 
                                    className="bg-blue-900 text-white text-xs font-semibold px-3 py-1.5 rounded-full flex items-center space-x-1.5 capitalize shadow-sm transition transform hover:scale-105"
                                >
                                    <span>{symptom}</span>
                                    <button 
                                        type="button" 
                                        onClick={() => handleToggleSymptom(symptom)} 
                                        className="ml-1 text-slate-300 hover:text-red-300 font-bold focus:outline-none"
                                        title="Remove symptom"
                                    >
                                        ✕
                                    </button>
                                </span>
                            ))
                        )}
                    </div>
                </div>

                {/* Category Browser & Interactive Selector Chips */}
                <div className="space-y-4 bg-slate-50 p-4 rounded-lg border border-slate-200">
                    {Object.entries(CATEGORIZED_SYMPTOMS).map(([category, symptoms]) => (
                        <div key={category}>
                            <h4 className="text-xs font-bold text-teal-800 uppercase tracking-wider mb-2">{category}</h4>
                            <div className="flex flex-wrap gap-2">
                                {symptoms.map((symptom) => {
                                    const isSelected = formData.symptoms.includes(symptom);
                                    return (
                                        <button
                                            key={symptom}
                                            type="button"
                                            onClick={() => handleToggleSymptom(symptom)}
                                            className={`text-xs font-medium px-3.5 py-1.5 rounded-full border capitalize transition-all duration-150 ${
                                                isSelected 
                                                    ? 'bg-teal-600 text-white border-teal-600 shadow-md transform scale-105' 
                                                    : 'bg-white text-slate-700 border-slate-300 hover:border-teal-500 hover:bg-teal-50 hover:text-teal-900'
                                            }`}
                                        >
                                            {isSelected ? '✓ ' : '+ '}{symptom}
                                        </button>
                                    );
                                })}
                            </div>
                        </div>
                    ))}
                </div>
            </div>

            {/* Submission Button */}
            <button 
                type="submit" 
                disabled={loading || formData.symptoms.length === 0} 
                className="w-full bg-blue-900 hover:bg-blue-800 text-white font-bold py-3 px-4 rounded-lg shadow-md hover:shadow-lg disabled:bg-slate-300 disabled:cursor-not-allowed transition duration-200"
            >
                {loading ? 'Evaluating Diagnosis...' : 'Run Clinical Diagnosis'}
            </button>
        </form>
    );
}
